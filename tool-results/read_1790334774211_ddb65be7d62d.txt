   579→function categoryLabelForSrc(src: string): string {

   581→  const entry = (Object.keys(GALLERY_BY_CATEGORY) as (keyof typeof GALLERY_BY_CATEGORY)[]).find(

   586→    .split('_')
   587→    .map((w) => w.charAt(0).toUpperCase() + w.slice(1))
   588→    .join(' ');
